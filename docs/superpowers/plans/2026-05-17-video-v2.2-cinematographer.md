**Goal:** Ship v2.2 of the video pipeline — Cinematographer agent with gemma 4 tool-call shim, cumulative-context Sourcer, real CTA-as-final-scene outro with silent 2s logo stinger, per-scene SSML prosody, and a persistent stealth Google Images browser.

**Architecture:** Insert a new `Cinematographer` agent stage after `Storyboarder` that emits per-scene cinematography decisions (color grade, transition, motion, voice prosody) via a forced-JSON tool-call shim. Composer, voiceover, and transition modules read those decisions instead of the hardcoded `mood → effect` maps (which become fallbacks). Sourcer gains a `NarrativeThread` (cumulative topical keywords per scene) so candidate scoring sees what the audience is holding in their head, not just the current line. Outro path in the composer is replaced with a real final scene rendered from a brand-logo template plus a silent 2s stinger. Voiceover renders per-segment SSML with prosody driven by the Cinematographer. OllamaClient is upgraded for Gemma 4 (thinking-mode + tool-call helper + sampling defaults). Google Images browser switches to `tf-playwright-stealth` with a persistent user-data-dir.

**Tech Stack:** Python 3.11+, pytest, MoviePy, ffmpeg, edge-tts (SSML mode), Ollama HTTP API (gemma 4), Playwright (sync API) + `tf-playwright-stealth`, Pillow, dataclasses.

```python
"""Thin wrapper around Ollama HTTP API with JSON-mode + tool-call helpers.

Gemma 4 specifics:
  * Sampling defaults: temperature=1.0, top_p=0.95, top_k=64 (per Google spec).
  * Thinking mode: enabled by prepending '<|think|>' to the system prompt;
    output begins with '<|channel>thought\\n...<channel|>' then the answer.
    We strip that block before returning.
  * Multi-turn rule: model output passed back as history must NOT include
    prior thought blocks. We don't currently use multi-turn — comment flag
    only.
"""
import json
import re
import logging
import requests
from video_agent.config import OLLAMA_HOST, OLLAMA_MODEL, OLLAMA_RETRY_MAX

log = logging.getLogger(__name__)


class OllamaError(RuntimeError):
    pass


_FENCE_RE = re.compile(r"^```(?:json)?\s*|\s*```$", re.IGNORECASE | re.MULTILINE)
_THOUGHT_RE = re.compile(
    r"<\|channel>thought\n.*?<channel\|>", re.DOTALL,
)

# Gemma 4 sampling defaults (Google spec, applied unless caller overrides).
_GEMMA4_DEFAULTS = {"temperature": 1.0, "top_p": 0.95, "top_k": 64}


class OllamaClient:
    def __init__(self, host: str = OLLAMA_HOST, model: str = OLLAMA_MODEL,
                 timeout: float | None = None, think_mode: bool = True):
        self.host = host
        self.model = model
        self.timeout = timeout
        self.think_mode = think_mode

    def generate(self, prompt: str, system: str | None = None,
                 temperature: float | None = None,
                 top_p: float | None = None,
                 top_k: int | None = None) -> str:
        """Send a single-turn generation. Strips Gemma 4 thought blocks
        from the response before returning."""
        opts = dict(_GEMMA4_DEFAULTS)
        if temperature is not None:
            opts["temperature"] = temperature
        if top_p is not None:
            opts["top_p"] = top_p
        if top_k is not None:
            opts["top_k"] = top_k

        sys_prompt = system or ""
        if self.think_mode:
            sys_prompt = "<|think|>" + sys_prompt

        body = {
            "model": self.model,
            "prompt": prompt,
            "stream": False,
            "options": opts,
        }
        if sys_prompt:
            body["system"] = sys_prompt

        try:
            r = requests.post(
                f"{self.host}/api/generate", json=body, timeout=self.timeout,
            )
            r.raise_for_status()
        except (requests.ConnectionError, ConnectionError) as e:
            raise OllamaError(
                f"Ollama not running at {self.host} — start with: ollama serve"
            ) from e
        except requests.Timeout as e:
            raise OllamaError(
                f"Ollama call timed out — model ({self.model}) took too long"
            ) from e
        except requests.HTTPError as e:
            raise OllamaError(f"Ollama HTTP error: {e}") from e

        raw = r.json().get("response", "")
        # Strip any Gemma 4 thought blocks before returning.
        stripped = _THOUGHT_RE.sub("", raw).strip()
        return stripped

    def generate_json(self, prompt: str, system: str | None = None,
                      retries: int = OLLAMA_RETRY_MAX) -> dict | list:
        sys = (system or "") + "\nRespond with raw JSON only. No prose, no markdown."
        last_err = None
        for attempt in range(1, retries + 1):
            raw = self.generate(prompt, system=sys)
            cleaned = _FENCE_RE.sub("", raw).strip()
            try:
                return json.loads(cleaned)
            except json.JSONDecodeError as e:
                last_err = e
                log.warning("Ollama JSON parse failed (attempt %d): %s", attempt, e)
        raise OllamaError(f"Ollama JSON parse failed after {retries} retries: {last_err}")

    def generate_tool_calls(self, prompt: str, system: str,
                            allowed_tools: set[str],
                            retries: int = OLLAMA_RETRY_MAX) -> list[dict]:
        """Forced-JSON tool-call shim for small models whose native function-
        calling is unreliable. Expects model to return:
            {"calls": [{"tool": "<name>", "args": {...}}, ...]}
        Returns the validated list of calls. Raises OllamaError on:
          - malformed JSON / missing 'calls' key
          - any call referencing a tool not in allowed_tools
          - any call missing 'tool' or 'args'
        """
        result = self.generate_json(prompt, system=system, retries=retries)
        if not isinstance(result, dict) or "calls" not in result:
            raise OllamaError(
                f"tool-call response missing 'calls' key: {result!r}"
            )
        calls = result["calls"]
        if not isinstance(calls, list):
            raise OllamaError(f"'calls' must be a list, got {type(calls)}")
        for c in calls:
            if not isinstance(c, dict) or "tool" not in c or "args" not in c:
                raise OllamaError(f"malformed call entry: {c!r}")
            if c["tool"] not in allowed_tools:
                raise OllamaError(
                    f"unknown tool {c['tool']!r}; allowed={sorted(allowed_tools)}"
                )
        return calls
```

```python
"""Cinematographer — emits per-scene cinematography decisions via a forced-JSON
tool-call shim on Gemma 4. Decisions land on Scene.cinematography. If the
tool-call fails or is malformed, every scene gets a default Cinematography
object so downstream consumers can always read its fields."""
from __future__ import annotations
import logging

from video_agent.ollama_client import OllamaClient, OllamaError
from video_agent.storyboard import Storyboard, Cinematography

log = logging.getLogger(__name__)

# Allowed values for each tool — anything outside is dropped silently.
_PALETTES = {"red_tension", "cold_blue", "warm_brand",
             "neutral_doc", "urgent_amber", "clinical_white"}
_TRANSITIONS = {"cut", "whip_pan", "dissolve_flash", "slow_fade", "hard_cut"}
_MOTIONS = {"slow_push", "hold", "fast_zoom", "drift", "parallax_lr"}
_PROSODIES = {"hook_emphasis", "urgent_problem", "conversational",
              "warm_cta", "matter_of_fact"}

_ALLOWED_TOOLS = {
    "set_color_grade", "set_transition", "set_motion", "set_voice_prosody",
}

_SYSTEM = """You are a Cinematographer for a 30-55s vertical B2B chemistry video.
For each scene, choose:
  - color_grade ∈ {red_tension, cold_blue, warm_brand, neutral_doc, urgent_amber, clinical_white}
  - transition_in ∈ {cut, whip_pan, dissolve_flash, slow_fade, hard_cut}
  - motion ∈ {slow_push, hold, fast_zoom, drift, parallax_lr}
  - voice_prosody ∈ {hook_emphasis, urgent_problem, conversational, warm_cta, matter_of_fact}

Guidance:
  * hook beat: hard_cut in, fast_zoom or parallax_lr, voice = hook_emphasis,
    color = red_tension or urgent_amber for a tension open.
  * mechanism beat: slow_push or hold, color = clinical_white or cold_blue,
    voice = conversational.
  * proof beat: dissolve_flash or slow_fade, voice = matter_of_fact.
  * cta beat: slow_fade in, motion = slow_push, color = warm_brand,
    voice = warm_cta.
  * Avoid back-to-back fast_zoom or back-to-back hard_cut.

Respond ONLY as JSON:
{"calls": [
  {"tool": "set_color_grade",   "args": {"scene": <int>, "palette": "<value>"}},
  {"tool": "set_transition",    "args": {"scene": <int>, "kind":    "<value>"}},
  {"tool": "set_motion",        "args": {"scene": <int>, "kind":    "<value>"}},
  {"tool": "set_voice_prosody", "args": {"scene": <int>, "tone":    "<value>"}}
]}
You SHOULD emit all four tool calls for every scene.
"""


class Cinematographer:
    def __init__(self, client: OllamaClient | None = None):
        self.client = client or OllamaClient()

    def run(self, sb: Storyboard) -> Storyboard:
        # First — every scene gets a default Cinematography so consumers can
        # always read its fields. Tool calls then refine.
        for s in sb.scenes:
            if s.cinematography is None:
                s.cinematography = Cinematography()

        prompt = self._build_prompt(sb)
        try:
            calls = self.client.generate_tool_calls(
                prompt, system=_SYSTEM, allowed_tools=_ALLOWED_TOOLS,
            )
        except OllamaError as e:
            log.warning("Cinematographer: tool-call failed (%s); "
                        "scenes will use default fallback cinematography", e)
            return sb

        applied = 0
        dropped = 0
        for c in calls:
            ok = self._apply_call(sb, c)
            if ok:
                applied += 1
            else:
                dropped += 1
        log.info("Cinematographer: applied=%d dropped=%d total_calls=%d",
                 applied, dropped, len(calls))
        return sb

    def _build_prompt(self, sb: Storyboard) -> str:
        hero = sb.hero_claim.claim_text if sb.hero_claim else ""
        lines = [
            f"Hero claim: {hero}",
            f"Region: {sb.blog.get('region')} | Category: {sb.blog.get('category')}",
            "Scenes:",
        ]
        for s in sb.scenes:
            lines.append(
                f"  scene {s.index} ({s.beat}, {s.duration_target_s:.1f}s): "
                f"{s.narration[:120]}"
            )
        lines.append("\nEmit cinematography tool calls for every scene.")
        return "\n".join(lines)

    def _apply_call(self, sb: Storyboard, c: dict) -> bool:
        args = c.get("args") or {}
        idx = args.get("scene")
        if not isinstance(idx, int) or idx < 0 or idx >= len(sb.scenes):
            return False
        cin = sb.scenes[idx].cinematography  # always non-None now
        tool = c["tool"]
        if tool == "set_color_grade":
            val = args.get("palette")
            if val in _PALETTES:
                cin.color_grade = val
                return True
        elif tool == "set_transition":
            val = args.get("kind")
            if val in _TRANSITIONS:
                cin.transition_in = val
                return True
        elif tool == "set_motion":
            val = args.get("kind")
            if val in _MOTIONS:
                cin.motion = val
                return True
        elif tool == "set_voice_prosody":
            val = args.get("tone")
            if val in _PROSODIES:
                cin.voice_prosody = val
                return True
        return False
```

```python
import logging
import subprocess
from pathlib import Path

log = logging.getLogger(__name__)

_PALETTE_FILTERS: dict[str, str] = {
    # 12% red shift + soft vignette — tension / problem open
    "red_tension": (
        "colorchannelmixer=rr=1.12:rg=0.0:rb=0.0:"
        "gr=0.0:gg=0.96:gb=0.0:br=0.0:bg=0.0:bb=0.96,"
        "vignette=PI/5"
    ),
    # 8% blue shift + 5% desaturation — sterile, mechanical feel
    "cold_blue": (
        "colorchannelmixer=rr=0.95:gg=0.97:bb=1.08,"
        "hue=s=0.95"
    ),
    # 6% warm tint + gentle vignette — conclusive / brand
    "warm_brand": (
        "colorchannelmixer=rr=1.06:gg=1.02:bb=0.94,"
        "vignette=PI/6:eval=init"
    ),
    # 10% amber + slight contrast bump — urgency without alarm
    "urgent_amber": (
        "colorchannelmixer=rr=1.10:gg=1.04:bb=0.90,"
        "eq=contrast=1.05"
    ),
    # Slight desat + +5% brightness — lab / clinical
    "clinical_white": (
        "hue=s=0.92,eq=brightness=0.05"
    ),
    # neutral_doc: explicitly no grade
    "neutral_doc": None,
}


def grade_filter_for_palette(palette: str) -> str | None:
    """Returns an ffmpeg filter chain for the palette, or None for no grading."""
    return _PALETTE_FILTERS.get(palette)


# Legacy mood-based map — used as fallback when scene has no Cinematography.
_MOOD_TO_PALETTE = {
    "problem": "red_tension",
    "brand": "warm_brand",
    "cta": "warm_brand",
}


def grade_filter_for_mood(mood: str) -> str | None:
    palette = _MOOD_TO_PALETTE.get(mood)
    if palette is None:
        return None
    return grade_filter_for_palette(palette)


def apply_grade(clip: Path, palette_or_mood: str) -> Path:
    """Renders an in-place graded copy; returns the same path.

    Accepts either a palette name (preferred) or a legacy mood ("problem",
    "brand"). Falls through to no-op when neither matches.
    On filter failure, leaves the clip untouched and logs a warning."""
    flt = grade_filter_for_palette(palette_or_mood)
    if flt is None:
        flt = grade_filter_for_mood(palette_or_mood)
    if not flt:
        return clip
    out = clip.with_name(clip.stem + "_graded.mp4")
    try:
        subprocess.run([
            "ffmpeg", "-y", "-loglevel", "error", "-i", str(clip),
            "-vf", flt, "-c:v", "libx264", "-preset", "fast",
            "-crf", "20", "-pix_fmt", "yuv420p", str(out),
        ], check=True)
    except subprocess.CalledProcessError as e:
        log.warning("apply_grade failed for %s (%s); leaving clip ungraded",
                    palette_or_mood, e)
        return clip
    clip.unlink()
    out.rename(clip)
    return clip
```

```python
def context_match_score(caption: str, narration: str,
                        thread_keywords: list[str] | None = None,
                        hero_claim: str | None = None) -> int:
    """Score a candidate's caption against the scene's narration plus the
    cumulative narrative thread (keywords from neighbouring scenes) and the
    hero claim. Cumulative context catches off-topic-but-on-word matches
    that a per-scene score misses.

    Returns an integer roughly in [0, 100]. 0 means hard-reject.
    Anything < 30 should not be used.
    """
    if not caption or not narration:
        return 0

    # Hard rejects FIRST
    if _CAPTION_REJECT.search(caption):
        return 0
    non_latin = len(_NON_LATIN.findall(caption))
    latin = len(_LATIN_LETTER.findall(caption))
    if non_latin > 0 and non_latin > latin * 0.15:
        return 0

    nar_toks = _tokens(narration)
    cap_toks = _tokens(caption)
    if not nar_toks or not cap_toks:
        return 10

    # Cumulative context: thread keywords + hero claim tokens
    thread_toks: set[str] = set()
    if thread_keywords:
        for kw in thread_keywords:
            thread_toks |= _tokens(kw)
    hero_toks = _tokens(hero_claim or "")
```

```python
"""Voiceover generation via edge-tts SSML (primary) with Kokoro-82M fallback.

Public surface:
  - VoiceSegment(text, prosody)          dataclass
  - synthesize_segments(segs, out, region) → dict  — preferred
  - synthesize(narration, out, region)   → dict    — back-compat shim

import asyncio
import logging
from dataclasses import dataclass
from pathlib import Path
from pydub import AudioSegment

import edge_tts

from video_agent.config import (
    TTS_VOICES, TTS_RATE, TTS_PITCH, KOKORO_DEFAULT_VOICE,
)
from video_agent.text_normalizer import normalize_for_tts

log = logging.getLogger(__name__)

MAX_WORDS = 250
MIN_DURATION_S = 30
MAX_DURATION_S = 65
MIN_FILE_BYTES = 1024
INTER_SEGMENT_GAP_MS = 120


@dataclass
class VoiceSegment:
    text: str
    prosody: str = "conversational"
    # hook_emphasis | urgent_problem | conversational | warm_cta | matter_of_fact


class VoiceoverError(RuntimeError):
    pass


_PROSODY_PRESETS = {
    "hook_emphasis":  {"rate": "-10%", "pitch": "+2st", "emphasis": True},
    "urgent_problem": {"rate": "+5%",  "pitch": "+1st", "emphasis": False},
    "conversational": {"rate": None,   "pitch": None,   "emphasis": False},
    "warm_cta":       {"rate": "-5%",  "pitch": "-1st", "emphasis": False},
    "matter_of_fact": {"rate": None,   "pitch": None,   "emphasis": False},
}


def _xml_escape(s: str) -> str:
    return (s.replace("&", "&amp;").replace("<", "&lt;")
             .replace(">", "&gt;").replace('"', "&quot;"))


def build_ssml_segment(text: str, prosody: str, voice: str) -> str:
    """Wrap text in an SSML document for a single edge-tts segment."""
    preset = _PROSODY_PRESETS.get(prosody, _PROSODY_PRESETS["conversational"])
    body = _xml_escape(text)
    if preset["emphasis"]:
        # Wrap the first ~4 words in <emphasis>
        words = body.split()
        head = " ".join(words[:4])
        tail = " ".join(words[4:])
        body = f"<emphasis level=\"strong\">{head}</emphasis>"
        if tail:
            body = f"{body} {tail}"
    attrs = []
    if preset["rate"]:
        attrs.append(f'rate="{preset["rate"]}"')
    if preset["pitch"]:
        attrs.append(f'pitch="{preset["pitch"]}"')
    if attrs:
        body = f"<prosody {' '.join(attrs)}>{body}</prosody>"
    return (
        '<speak version="1.0" xmlns="http://www.w3.org/2001/10/synthesis" '
        'xml:lang="en-US">'
        f'<voice name="{voice}">{body}</voice></speak>'
    )


def _edge_synthesize_ssml(ssml: str, voice: str, output_path: Path) -> None:
    async def _run():
        comm = edge_tts.Communicate(ssml, voice=voice)
        await comm.save(str(output_path))
    try:
        asyncio.run(_run())
    except RuntimeError as e:
        if "cannot be called from a running event loop" in str(e):
            loop = asyncio.get_event_loop()
            loop.run_until_complete(_run())
        else:
            raise


def _kokoro_synthesize(text: str, output_path: Path) -> None:
    try:
        from kokoro_onnx import Kokoro
    except ImportError as e:
        raise VoiceoverError(
            "Kokoro fallback requires kokoro-onnx — pip install kokoro-onnx"
        ) from e
    kokoro = Kokoro.from_pretrained()
    samples, sr = kokoro.create(text, voice=KOKORO_DEFAULT_VOICE)
    seg = AudioSegment(
        samples.tobytes(), frame_rate=sr,
        sample_width=samples.dtype.itemsize, channels=1,
    )
    seg.export(output_path, format="mp3", bitrate="128k")


def synthesize_segments(segments: list[VoiceSegment], output_path: Path,
                        region: str,
                        voice_override: str | None = None) -> dict:
    """Render per-segment SSML through edge-tts, concatenate with a small gap.
    Returns the same result-dict shape as the legacy synthesize()."""
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    voice = voice_override or TTS_VOICES.get(region, TTS_VOICES["default"])

    fell_back = False
    engine = "edge-tts-ssml"
    pieces: list[AudioSegment] = []

    for i, seg in enumerate(segments):
        normalized = normalize_for_tts(seg.text)
        if not normalized.strip():
            continue
        seg_path = output_path.with_name(
            output_path.stem + f"_seg{i:02d}.mp3"
        )
        try:
            ssml = build_ssml_segment(normalized, seg.prosody, voice)
            _edge_synthesize_ssml(ssml, voice, seg_path)
            if not seg_path.exists() or seg_path.stat().st_size < MIN_FILE_BYTES:
                raise RuntimeError("edge-tts SSML output too small")
        except Exception as e:
            log.warning("edge-tts SSML failed for seg %d (%s) — Kokoro fallback",
                        i, e)
            fell_back = True
            engine = "kokoro"
            _kokoro_synthesize(normalized, seg_path)
        pieces.append(AudioSegment.from_mp3(str(seg_path)))
        seg_path.unlink(missing_ok=True)

    if not pieces:
        raise VoiceoverError("no segments produced any audio")

    final = pieces[0]
    gap = AudioSegment.silent(duration=INTER_SEGMENT_GAP_MS)
    for p in pieces[1:]:
        final = final + gap + p
    final.export(str(output_path), format="mp3", bitrate="192k")

    duration_s = len(final) / 1000.0
    if not (MIN_DURATION_S <= duration_s <= MAX_DURATION_S):
        log.warning("Voiceover duration %.1fs outside target [%d, %d]",
                    duration_s, MIN_DURATION_S, MAX_DURATION_S)

    return {
        "audio_path": output_path,
        "duration_s": duration_s,
        "voice_used": voice,
        "engine_used": engine,
        "fell_back": fell_back,
    }

               voice_override: str | None = None) -> dict:
    """Back-compat shim: render the whole narration as a single conversational
    segment. Prefer synthesize_segments() for per-scene prosody."""

                    len(words), MAX_WORDS)

        [VoiceSegment(text=narration, prosody="conversational")],
```

```python
"""Brand-template renderer for the final CTA scene and the silent 2s
logo stinger that closes the video.

These replace the deprecated pre-rendered OUTRO_VIDEO_PATH + bolted-on
templated CTA voiceover path."""
from __future__ import annotations
import subprocess
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from video_agent.config import BRAND_NAVY_2
from video_agent.safezone import FRAME_W, FRAME_H

STINGER_DURATION_S = 2.0


def _compose_brand_still(logo_path: Path, url_text: str,
                          out_jpg: Path) -> Path:
    """Compose a single 1080x1920 PNG with brand-navy background, logo
    in the top-third, URL wordmark in the bottom-third."""
    bg = Image.new("RGB", (FRAME_W, FRAME_H), BRAND_NAVY_2)

    # Logo, centered top-third, scaled to ~50% of frame width
    logo = Image.open(logo_path).convert("RGBA")
    target_w = int(FRAME_W * 0.50)
    aspect = logo.height / max(1, logo.width)
    logo_resized = logo.resize((target_w, int(target_w * aspect)),
                               Image.LANCZOS)
    lx = (FRAME_W - logo_resized.width) // 2
    ly = int(FRAME_H * 0.28)
    bg.paste(logo_resized, (lx, ly), logo_resized)

    # URL wordmark — bottom-third, clean type
    draw = ImageDraw.Draw(bg)
    try:
        font = ImageFont.truetype("Poppins-Bold.ttf", 78)
    except OSError:
        font = ImageFont.load_default()
    tw = draw.textlength(url_text, font=font)
    tx = (FRAME_W - tw) // 2
    ty = int(FRAME_H * 0.72)
    draw.text((tx, ty), url_text, fill=(255, 255, 255), font=font)

    out_jpg.parent.mkdir(parents=True, exist_ok=True)
    bg.save(out_jpg, "PNG")
    return out_jpg


def render_cta_scene(logo_path: Path, url_text: str,
                      duration_s: float, output_path: Path,
                      fps: int = 30) -> Path:
    """Render the final CTA scene: brand background + logo settles in
    (scale 1.05 → 1.0, opacity 0 → 1 over first 0.6s) + URL wordmark.

    Output is video-only (no audio). The voiceover is added by the
    composer in the main mux pass."""
    logo_path = Path(logo_path)
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    still = output_path.with_suffix(".still.png")
    _compose_brand_still(logo_path, url_text, still)

    # Ken-Burns-style settle: fade in 0.6s + slight zoom-out
    total_frames = max(1, int(duration_s * fps))
    settle_frames = max(1, int(0.6 * fps))
    # zoompan: zoom from 1.05 to 1.00 over settle frames, then hold
    zoom_expr = f"'if(lte(on,{settle_frames}),1.05-0.05*on/{settle_frames},1.0)'"
    vf = (
        f"zoompan=z={zoom_expr}:d={total_frames}:s={FRAME_W}x{FRAME_H}:fps={fps},"
        f"fade=t=in:st=0:d=0.6,setsar=1"
    )
    subprocess.run([
        "ffmpeg", "-y", "-loglevel", "error",
        "-loop", "1", "-i", str(still),
        "-vf", vf, "-t", str(duration_s), "-r", str(fps),
        "-c:v", "libx264", "-preset", "fast", "-crf", "20",
        "-pix_fmt", "yuv420p", "-an", str(output_path),
    ], check=True)
    still.unlink(missing_ok=True)
    return output_path


def render_logo_stinger(logo_path: Path, output_path: Path,
                         fps: int = 30) -> Path:
    """Render a silent 2-second logo stinger: held brand still that
    crossfades in over the first 0.4s. Closes the video."""
    still = output_path.with_suffix(".stinger_still.png")
    _compose_brand_still(logo_path, "hrsuindore.com", still)
    total_frames = max(1, int(STINGER_DURATION_S * fps))
    vf = (
        f"fade=t=in:st=0:d=0.4,setsar=1,scale={FRAME_W}:{FRAME_H},"
        f"trim=duration={STINGER_DURATION_S}"
    )
    subprocess.run([
        "ffmpeg", "-y", "-loglevel", "error",
        "-loop", "1", "-i", str(still),
        "-vf", vf, "-t", str(STINGER_DURATION_S), "-r", str(fps),
        "-c:v", "libx264", "-preset", "fast", "-crf", "20",
        "-pix_fmt", "yuv420p", "-an", str(output_path),
    ], check=True)
    still.unlink(missing_ok=True)
    return output_path
```

```python
class GoogleImagesBrowserSource(BaseSource):
    name = "google_images"
    authority_weight = 5

    def __init__(self):
        self._executor = ThreadPoolExecutor(
            max_workers=1, thread_name_prefix="gimg-pw")
        self._pw = None
        self._ctx = None       # BrowserContext (persistent)
        self._page = None      # Long-lived single page reused across queries
        self._init_failed = False
        try:
            self._executor.submit(self._init_browser).result(timeout=20)
        except Exception as e:
            log.warning("GoogleImages: init dispatch failed (%s)", e)
            self._init_failed = True
        atexit.register(self._shutdown)

    # ---- runs ONLY in the executor thread ----
    def _init_browser(self):
        try:
            self._pw = sync_playwright().start()
            _PROFILE_DIR.mkdir(parents=True, exist_ok=True)
            self._ctx = self._pw.chromium.launch_persistent_context(
                user_data_dir=str(_PROFILE_DIR.resolve()),
                headless=not _INTERACTIVE,
                user_agent=_UA,
                viewport={"width": 1366, "height": 900},
                locale="en-US",
                args=[
                    "--disable-blink-features=AutomationControlled",
                    "--no-sandbox",
                    "--disable-dev-shm-usage",
                ],
            )
            # Apply the stealth fingerprint patches to the context
            try:
                Stealth().apply_stealth_sync(self._ctx)
            except Exception as e:
                log.warning("Stealth().apply_stealth_sync failed (%s); "
                            "continuing without stealth patches", e)
            # Reuse the first page Chromium opens with a persistent profile;
            # otherwise create one.
            if self._ctx.pages:
                self._page = self._ctx.pages[0]
            else:
                self._page = self._ctx.new_page()
            if _INTERACTIVE:
                log.info("GoogleImages: INTERACTIVE — visible browser, "
                         "manual CAPTCHA solve if hit")
            log.info("GoogleImages: persistent profile loaded at %s",
                     _PROFILE_DIR)
        except Exception as e:
            log.warning("GoogleImages: Playwright init failed (%s); "
                        "this source returns []", e)
            self._init_failed = True

    # ---- runs ONLY in the executor thread ----
    def _do_search(self, query: str, limit: int) -> list[RawCandidate]:
        if self._init_failed or self._page is None:
            return []
        try:
            self._page.goto(
                f"https://www.google.com/search?q={quote_plus(query)}"
                f"&tbm=isch&hl=en&safe=active",
                timeout=15000, wait_until="domcontentloaded",
            )
            if "/sorry/" in self._page.url or "captcha" in self._page.url.lower():
                if _INTERACTIVE:
                    print("\n" + "=" * 70, file=sys.stderr, flush=True)
                    print(f">>> Google CAPTCHA hit for query: {query!r}",
                          file=sys.stderr, flush=True)
                    print(">>> Solve it in the Chromium window. Persistent "
                          "profile means you should see this MUCH less often "
                          "from now on.", file=sys.stderr, flush=True)
                    print(">>> Press ENTER here when results are visible.",
                          file=sys.stderr, flush=True)
                    print("=" * 70, file=sys.stderr, flush=True)
                    try:
                        input()
                    except EOFError:
                        pass
                    if "/sorry/" in self._page.url:
                        try:
                            self._page.goto(
                                f"https://www.google.com/search?q="
                                f"{quote_plus(query)}&tbm=isch&hl=en",
                                timeout=20000, wait_until="domcontentloaded",
                            )
                            self._page.wait_for_timeout(1500)
                        except Exception as e:
                            log.warning("post-CAPTCHA reload failed: %s", e)
                            return []
                    if "/sorry/" in self._page.url:
                        log.warning("GoogleImages: still on CAPTCHA — giving "
                                    "up on %r", query)
                        return []
                    log.info("GoogleImages: CAPTCHA cleared, continuing")
                else:
                    log.warning(
                        "GoogleImages: CAPTCHA for %r — set "
                        "GOOGLE_IMAGES_INTERACTIVE=1 to solve manually.", query,
                    )
                    return []
            self._page.wait_for_timeout(1200)
            try:
                self._page.wait_for_selector("img", timeout=4000)
            except Exception:
                pass
            _human_browse(self._page)
            data = self._page.evaluate(_EXTRACT_JS)
        except Exception as e:
            log.warning("GoogleImages: page interaction failed for %r: %s",
                        query, e)
            data = []

        originals: list[RawCandidate] = []
        thumbs: list[RawCandidate] = []
        seen: set[str] = set()
        for d in data:
            src = d.get("src", "")
            original = d.get("original", "") or ""
            w = int(d.get("w") or 0)
            h = int(d.get("h") or 0)
            alt = d.get("alt", "") or ""
            url = original if original else src
            if not url or url in seen:
                continue
            seen.add(url)
            is_thumb = any(s in url for s in _SKIP_HOSTS)
            cand = RawCandidate(
                source=self.name, url=url, caption=alt,
                width=w, height=h,
            )
            (thumbs if is_thumb else originals).append(cand)

        results = (originals or thumbs)[:limit]
        if results:
            log.info("GoogleImages: %d candidates for %r (%s)",
                     len(results), query,
                     "originals" if originals else "thumbs-only fallback")
        else:
            log.warning("GoogleImages: 0 candidates for %r", query)
        return results

    # ---- public API: any thread can call this ----
    def search(self, query: str, limit: int = 5) -> list[RawCandidate]:
        if self._init_failed:
            return []
        try:
            timeout = None if _INTERACTIVE else 25
            return self._executor.submit(
                self._do_search, query, limit
            ).result(timeout=timeout)
        except Exception as e:
            log.warning("GoogleImages: search dispatch failed for %r: %s",
                        query, e)
            return []

    def _shutdown(self):
        def _close():
            try:
                if self._page:
                    try:
                        self._page.close()
                    except Exception:
                        pass
                if self._ctx:
                    self._ctx.close()
                if self._pw:
                    self._pw.stop()
            except Exception:
                pass
        try:
            self._executor.submit(_close).result(timeout=5)
        except Exception:
            pass
        try:
            self._executor.shutdown(wait=False)
        except Exception:
            pass
```

```python
def _write_quality_report(sb, workspace: Path, voice_duration: float,
                          video_duration: float, outro_concatenated: bool) -> None:
    """Writes a machine-readable quality_report.json next to the final video."""
    import json
    from dataclasses import asdict

    report = {
        "video_duration_s": round(video_duration, 2),
        "voice_duration_s": round(voice_duration, 2),
        "durations_redistributed": True,
        "intro_concatenated": False,
        "outro_concatenated": outro_concatenated,
        "narrative_thread": list(sb.narrative_thread or []),
        "scenes": [
            {
                "index": s.index,
                "mood": getattr(s.visual_concept, "mood", None),
                "duration_s": round(s.duration_target_s, 2),
                "source": (s.chosen_asset.source if s.chosen_asset else None),
                "score": (s.chosen_asset.score if s.chosen_asset else None),
                "width": (s.chosen_asset.width if s.chosen_asset else 0),
                "height": (s.chosen_asset.height if s.chosen_asset else 0),
                "critic_flags": list(getattr(s.critic_notes, "flags", None) or []),
                "re_sourced": getattr(s, "_re_sourced", False),
                "degraded": s.degraded,
                "cinematography": (asdict(s.cinematography)
                                    if s.cinematography else None),
            } for s in sb.scenes
        ],
    }
    output_path = workspace / "quality_report.json"
    output_path.write_text(json.dumps(report, indent=2), encoding="utf-8")
```
